# 1. 获取登录页面，解析加密密钥 login_url = BASE_SSO + "/login?service=" + requests.utils.quote(SVC, safe="") r = session.get(login_url, timeout=15) fields = parse_fields(r.text) ck = base64.b64decode(fields["login-croypto"]) # 2. 加密密码 def enc(text, key): c = AES.new(key, AES.MODE_ECB) return base64.b64encode(c.encrypt(pad(text.encode(), 16))).decode() # 3. 提交登录获取Ticket data = [ ("username", USERNAME), ("password", enc(PASSWORD, ck)), # 加密后的密码 # ... 其他字段 ] r = session.post(BASE_SSO + "/login", data=data, allow_redirects=False) ticket = re.search(r"ticket=([^&\s]+)", r.headers["Location"]).group(1)

Cookie机制

# 从响应中提取关键Cookie weu_list, jsessionid = [], None for resp_obj in r.history + [r]: for c in resp_obj.headers.get("Set-Cookie", "").split(","): m = re.search(r"_WEU=([^;]+)", c) if m: weu_list.append(m.group(1)) m = re.search(r"JSESSIONID=([^;]+)", c) if m: jsessionid = m.group(1) # 后续请求使用Cookie "Cookie": f"_WEU={session._WEU};EMAP_LANG=zh; JSESSIONID={session._JSESSIONID}"
